cmake_minimum_required(VERSION 3.26)
project(CyberDesktopPublicPlugins LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Public plugins currently require Windows x64 MSVC /MT")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(CD_CLOCK_BUILD_TESTS "Build the optional clock developer tests" OFF)
option(CD_WIDGET_BUILD_TESTS "Build the optional widget developer tests" OFF)
option(CD_MONITOR_BUILD_TESTS "Build the optional system-info developer tests" OFF)
set(CD_PLUGINS_SDK_MODE INSTALLED CACHE STRING "SDK dependency: INSTALLED or SOURCE")
set_property(CACHE CD_PLUGINS_SDK_MODE PROPERTY STRINGS INSTALLED SOURCE)
set(CD_PLUGINS_SDK_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../sdk" CACHE PATH "SDK source for SOURCE mode")
option(CD_PLUGINS_SDK_ONLY "Build the SDK without any plugins (SOURCE mode)" OFF)
set(CD_PLUGIN_FAMILIES "ALL" CACHE STRING "Plugin families or ALL; dependencies are automatic")
set(CD_PLUGIN_IDS "" CACHE STRING "Optional enabled plugin IDs instead of families")


if(CD_PLUGINS_SDK_MODE STREQUAL "SOURCE")
    set(CD_SDK_BUILD_TOOLS ON CACHE BOOL "Plugin packaging needs the SDK tool")
    set(CD_SDK_BUILD_EXAMPLES OFF CACHE BOOL "Examples are built separately")
    set(CD_SDK_BUILD_TESTS OFF CACHE BOOL "Contract tests are built separately")
    include("${CD_PLUGINS_SDK_SOURCE}/cmake/NativeCompilerDependencies.cmake")
    add_subdirectory("${CD_PLUGINS_SDK_SOURCE}" sdk)
elseif(CD_PLUGINS_SDK_MODE STREQUAL "INSTALLED")
    if(CD_PLUGINS_SDK_ONLY)
        message(FATAL_ERROR "SDK-only builds require CD_PLUGINS_SDK_MODE=SOURCE")
    endif()
    find_package(CyberDesktopSDK 1 CONFIG REQUIRED COMPONENTS SDK)
else()
    message(FATAL_ERROR "Unknown SDK mode: ${CD_PLUGINS_SDK_MODE}")
endif()
if(CD_PLUGINS_SDK_ONLY)
    return()
endif()

include("${CMAKE_CURRENT_SOURCE_DIR}/common/Selection.cmake")
if(TARGET cd_brick_clock_core AND TARGET cd_marble_clock_core AND TARGET cd_precision_clocks_core)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../tests/render-cache" render-cache-tests)
endif()
