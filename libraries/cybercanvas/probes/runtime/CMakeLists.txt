cmake_minimum_required(VERSION 3.21)
project(RenderingRuntimeBenchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
find_package(CyberCanvas 1.2 CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
add_executable(rendering_runtime_benchmark RuntimeBenchmark.cpp)
target_link_libraries(rendering_runtime_benchmark PRIVATE CyberCanvas::Physics2D CyberCanvas::Effects
  CyberCanvas::Particles CyberCanvas::Layout CyberCanvas::Vector nlohmann_json::nlohmann_json)
target_compile_options(rendering_runtime_benchmark PRIVATE /utf-8 /W4)
add_executable(rendering_text_acceptance TextAcceptance.cpp)
target_link_libraries(rendering_text_acceptance PRIVATE CyberCanvas::Text)
target_compile_options(rendering_text_acceptance PRIVATE /utf-8 /W4)
add_executable(rendering_physics_acceptance PhysicsAcceptance.cpp)
target_link_libraries(rendering_physics_acceptance PRIVATE CyberCanvas::Physics2D)
target_compile_options(rendering_physics_acceptance PRIVATE /utf-8 /W4)
