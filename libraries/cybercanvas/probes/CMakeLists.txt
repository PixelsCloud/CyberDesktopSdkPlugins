cmake_minimum_required(VERSION 3.22)
project(CyberCanvasProbes LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
find_package(box2d CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_path(THORVG_INCLUDE_DIR thorvg.h PATH_SUFFIXES thorvg-1 REQUIRED)
find_library(THORVG_LIBRARY NAMES thorvg-1 thorvg REQUIRED)
add_executable(rendering_probe RenderingProbe.cpp)
target_compile_definitions(rendering_probe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE TVG_STATIC)
target_compile_options(rendering_probe PRIVATE /W4 /utf-8)
target_include_directories(rendering_probe PRIVATE "${THORVG_INCLUDE_DIR}" ../include)
target_link_libraries(rendering_probe PRIVATE box2d::box2d nlohmann_json::nlohmann_json
    "${THORVG_LIBRARY}" d2d1 dwrite d3d11 dxgi dxguid windowscodecs ole32)
