# Private implementations and test support use paths rooted at this library.
# Only include/CyberCanvas remains part of the installed public interface.
get_property(canvas_local_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
foreach(target IN LISTS canvas_local_targets)
    get_target_property(type ${target} TYPE)
    if(type STREQUAL "STATIC_LIBRARY" OR type STREQUAL "SHARED_LIBRARY" OR type STREQUAL "EXECUTABLE")
        target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    endif()
endforeach()
