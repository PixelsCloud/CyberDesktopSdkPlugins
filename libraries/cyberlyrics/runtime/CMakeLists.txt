cmake_minimum_required(VERSION 3.24)
project(CyberLyricsRuntime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CYBERLYRICS_FFMPEG_ROOT "" CACHE PATH "Audited static LGPL FFmpeg installation prefix")
if(NOT EXISTS "${CYBERLYRICS_FFMPEG_ROOT}/share/ffmpeg/FindFFMPEG.cmake")
    message(FATAL_ERROR "Set CYBERLYRICS_FFMPEG_ROOT to the LGPL FFmpeg installation")
endif()
include("${CYBERLYRICS_FFMPEG_ROOT}/share/ffmpeg/FindFFMPEG.cmake")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/Runtime.cpp")
    set(lyrics_root "${CMAKE_CURRENT_SOURCE_DIR}")
else()
    get_filename_component(lyrics_root "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
endif()
add_library(cyberlyrics_runtime SHARED "${lyrics_root}/src/Runtime.cpp")
target_include_directories(cyberlyrics_runtime PRIVATE "${lyrics_root}/include" ${FFMPEG_INCLUDE_DIRS})
target_compile_definitions(cyberlyrics_runtime PRIVATE CYBERLYRICS_RUNTIME_BUILD NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(cyberlyrics_runtime PRIVATE ${FFMPEG_LIBRARIES})
if(MSVC)
    target_compile_options(cyberlyrics_runtime PRIVATE /utf-8 /W4)
endif()
