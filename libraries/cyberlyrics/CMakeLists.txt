cmake_minimum_required(VERSION 3.24)
project(CyberLyrics VERSION 0.1.0 LANGUAGES CXX)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(CYBERLYRICS_BUILD_TESTS "Build focused lyrics tests" ${PROJECT_IS_TOP_LEVEL})
option(CYBERLYRICS_WITH_PARSER "Build the replaceable LGPL FFmpeg parser" ON)
option(CYBERLYRICS_WITH_WINDOWS_MEDIA "Build native Windows media adaptation" ${WIN32})
option(CYBERLYRICS_WITH_QT_VIEW "Build the shared shaped-text view adapter" ON)
option(CYBERLYRICS_WITH_WIDGET_VIEW "Build the reusable optional Qt Widgets lyrics view" OFF)
add_library(cyberlyrics_core STATIC src/Document.cpp src/Codec.cpp)
add_library(CyberLyrics::Core ALIAS cyberlyrics_core)
set_target_properties(cyberlyrics_core PROPERTIES EXPORT_NAME Core AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
target_compile_features(cyberlyrics_core PUBLIC cxx_std_20)
target_link_libraries(cyberlyrics_core PUBLIC Qt6::Core)
target_include_directories(cyberlyrics_core PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
if(MSVC)
    target_compile_options(cyberlyrics_core PRIVATE /utf-8 /W4)
endif()
install(TARGETS cyberlyrics_core EXPORT CyberLyricsTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
if(CYBERLYRICS_WITH_QT_VIEW)
    find_package(Qt6 6.5 REQUIRED COMPONENTS Gui)
    add_library(cyberlyrics_qt STATIC src/LyricLayout.cpp)
    add_library(CyberLyrics::Qt ALIAS cyberlyrics_qt)
    target_link_libraries(cyberlyrics_qt PUBLIC CyberLyrics::Core Qt6::Gui)
    set_target_properties(cyberlyrics_qt PROPERTIES EXPORT_NAME Qt AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
    if(MSVC)
        target_compile_options(cyberlyrics_qt PRIVATE /utf-8 /W4)
    endif()
    install(TARGETS cyberlyrics_qt EXPORT CyberLyricsTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()
if(CYBERLYRICS_WITH_WIDGET_VIEW)
    if(NOT CYBERLYRICS_WITH_QT_VIEW)
        message(FATAL_ERROR "The lyrics widget requires CYBERLYRICS_WITH_QT_VIEW")
    endif()
    find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets)
    add_library(cyberlyrics_widgets STATIC src/LyricsView.cpp src/ViewAnimation.cpp include/CyberLyrics/LyricsView.h)
    add_library(CyberLyrics::Widgets ALIAS cyberlyrics_widgets)
    target_link_libraries(cyberlyrics_widgets PUBLIC CyberLyrics::Qt Qt6::Widgets)
    set_target_properties(cyberlyrics_widgets PROPERTIES EXPORT_NAME Widgets AUTOMOC ON AUTOUIC OFF AUTORCC OFF)
    if(MSVC)
        target_compile_definitions(cyberlyrics_widgets PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
        target_compile_options(cyberlyrics_widgets PRIVATE /utf-8 /W4)
    endif()
    if(WIN32)
        target_link_libraries(cyberlyrics_widgets PRIVATE user32)
    endif()
    install(TARGETS cyberlyrics_widgets EXPORT CyberLyricsTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()
if(CYBERLYRICS_WITH_WINDOWS_MEDIA)
    find_package(Qt6 6.5 REQUIRED COMPONENTS Gui)
    add_library(cyberlyrics_windows_media STATIC src/WindowsMedia.cpp src/MediaTimelineClock.cpp)
    add_library(CyberLyrics::WindowsMedia ALIAS cyberlyrics_windows_media)
    target_link_libraries(cyberlyrics_windows_media PUBLIC CyberLyrics::Core Qt6::Gui PRIVATE windowsapp)
    set_target_properties(cyberlyrics_windows_media PROPERTIES EXPORT_NAME WindowsMedia AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
    target_compile_definitions(cyberlyrics_windows_media PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    if(MSVC)
        target_compile_options(cyberlyrics_windows_media PRIVATE /utf-8 /W4)
    endif()
    install(TARGETS cyberlyrics_windows_media EXPORT CyberLyricsTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()
if(CYBERLYRICS_WITH_PARSER)
    set(CYBERLYRICS_FFMPEG_ROOT "" CACHE PATH "Audited LGPL FFmpeg prefix")
    if(NOT EXISTS "${CYBERLYRICS_FFMPEG_ROOT}/share/ffmpeg/FindFFMPEG.cmake")
        message(FATAL_ERROR "Set CYBERLYRICS_FFMPEG_ROOT to the audited LGPL FFmpeg installation")
    endif()
    include("${CYBERLYRICS_FFMPEG_ROOT}/share/ffmpeg/FindFFMPEG.cmake")
    add_library(cyberlyrics_runtime SHARED src/Runtime.cpp)
    target_compile_features(cyberlyrics_runtime PRIVATE cxx_std_20)
    target_compile_definitions(cyberlyrics_runtime PRIVATE CYBERLYRICS_RUNTIME_BUILD)
    target_include_directories(cyberlyrics_runtime PRIVATE include ${FFMPEG_INCLUDE_DIRS})
    target_link_libraries(cyberlyrics_runtime PRIVATE ${FFMPEG_LIBRARIES})
    set_target_properties(cyberlyrics_runtime PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    add_library(cyberlyrics_parser STATIC src/Parser.cpp)
    add_library(CyberLyrics::Parser ALIAS cyberlyrics_parser)
    set_target_properties(cyberlyrics_parser PROPERTIES EXPORT_NAME Parser AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
    target_link_libraries(cyberlyrics_parser PUBLIC CyberLyrics::Core)
    add_dependencies(cyberlyrics_parser cyberlyrics_runtime)
    if(MSVC)
        target_compile_definitions(cyberlyrics_parser PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
        target_compile_options(cyberlyrics_parser PRIVATE /utf-8 /W4)
        target_compile_options(cyberlyrics_runtime PRIVATE /utf-8 /W4)
    endif()
    install(TARGETS cyberlyrics_parser cyberlyrics_runtime EXPORT CyberLyricsTargets
        ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
        LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
    foreach(port ffmpeg dav1d)
        install(FILES "${CYBERLYRICS_FFMPEG_ROOT}/share/${port}/copyright"
            DESTINATION "${CMAKE_INSTALL_DATADIR}/CyberLyrics/licenses/${port}")
    endforeach()
    find_package(cpr CONFIG REQUIRED)
    add_library(cyberlyrics_engine STATIC src/Engine.cpp src/Providers.cpp)
    add_library(CyberLyrics::Engine ALIAS cyberlyrics_engine)
    target_link_libraries(cyberlyrics_engine PUBLIC CyberLyrics::Core PRIVATE CyberLyrics::Parser cpr::cpr)
    set_target_properties(cyberlyrics_engine PROPERTIES EXPORT_NAME Engine AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
    if(MSVC)
        target_compile_options(cyberlyrics_engine PRIVATE /utf-8 /W4)
    endif()
    install(TARGETS cyberlyrics_engine EXPORT CyberLyricsTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../../LICENSE"
    DESTINATION "${CMAKE_INSTALL_DATADIR}/CyberLyrics/licenses" RENAME MIT.txt)
install(EXPORT CyberLyricsTargets NAMESPACE CyberLyrics:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/CyberLyrics)
configure_package_config_file(cmake/CyberLyricsConfig.cmake.in
    "${CMAKE_CURRENT_BINARY_DIR}/CyberLyricsConfig.cmake"
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/CyberLyrics)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/CyberLyricsConfigVersion.cmake"
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/CyberLyricsConfig.cmake"
    "${CMAKE_CURRENT_BINARY_DIR}/CyberLyricsConfigVersion.cmake"
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/CyberLyrics)
if(CYBERLYRICS_BUILD_TESTS)
    enable_testing()
    if(CYBERLYRICS_WITH_WIDGET_VIEW)
        add_executable(cyberlyrics_view_tests EXCLUDE_FROM_ALL tests/LyricsViewTest.cpp)
        target_link_libraries(cyberlyrics_view_tests PRIVATE CyberLyrics::Widgets)
        if(MSVC)
            target_compile_options(cyberlyrics_view_tests PRIVATE /utf-8 /W4)
        endif()
        add_test(NAME CyberLyricsView COMMAND cyberlyrics_view_tests)
        set_tests_properties(CyberLyricsView PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
    endif()
    add_executable(cyberlyrics_tests tests/CoreTest.cpp)
    target_link_libraries(cyberlyrics_tests PRIVATE CyberLyrics::Core)
    if(MSVC)
        target_compile_options(cyberlyrics_tests PRIVATE /utf-8 /W4)
    endif()
    add_test(NAME CyberLyricsCore COMMAND cyberlyrics_tests)
    if(CYBERLYRICS_WITH_PARSER)
        add_executable(cyberlyrics_api_probe EXCLUDE_FROM_ALL tests/ApiProbe.cpp)
        target_link_libraries(cyberlyrics_api_probe PRIVATE CyberLyrics::Engine)
        if(MSVC)
            target_compile_options(cyberlyrics_api_probe PRIVATE /utf-8 /W4)
        endif()
        add_executable(cyberlyrics_parser_engine_tests tests/ParserEngineTest.cpp)
        set_target_properties(cyberlyrics_parser_engine_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
        target_link_libraries(cyberlyrics_parser_engine_tests PRIVATE CyberLyrics::Parser CyberLyrics::Engine cyberlyrics_runtime)
        if(MSVC)
            target_compile_options(cyberlyrics_parser_engine_tests PRIVATE /utf-8 /W4)
        endif()
        add_test(NAME CyberLyricsParserEngine COMMAND cyberlyrics_parser_engine_tests)
        if(CYBERLYRICS_WITH_WINDOWS_MEDIA)
            add_executable(cyberlyrics_media_probe EXCLUDE_FROM_ALL tests/WindowsMediaProbe.cpp)
            target_link_libraries(cyberlyrics_media_probe PRIVATE CyberLyrics::Engine CyberLyrics::WindowsMedia)
            if(MSVC)
                target_compile_options(cyberlyrics_media_probe PRIVATE /utf-8 /W4)
            endif()
        endif()
    endif()
endif()
