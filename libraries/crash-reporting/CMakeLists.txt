target_compile_options(cyber_crash_reporting PRIVATE /utf-8 /W4)
target_compile_definitions(cyber_crash_reporting PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
set_target_properties(cyber_crash_reporting PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF MSVC_RUNTIME_LIBRARY MultiThreaded)

function(cyber_enable_crash_reporting directory)
    get_property(targets DIRECTORY "${directory}" PROPERTY BUILDSYSTEM_TARGETS)
    foreach(target IN LISTS targets)
        get_target_property(type ${target} TYPE)
        if(type STREQUAL "EXECUTABLE" AND (target MATCHES "^cyber(_|desktop_)" OR target STREQUAL "cyberimage"))
            get_target_property(enabled ${target} CYBER_CRASH_ENABLED)
            if(NOT enabled)
                target_sources(${target} PRIVATE "${CMAKE_CURRENT_FUNCTION_LIST_DIR}/Bootstrap.cpp")
                target_link_libraries(${target} PRIVATE cyber_crash_reporting)
                set_property(TARGET ${target} PROPERTY CYBER_CRASH_ENABLED TRUE)
            endif()
        endif()
    endforeach()
    get_property(children DIRECTORY "${directory}" PROPERTY SUBDIRECTORIES)
    foreach(child IN LISTS children)
        cyber_enable_crash_reporting("${child}")
    endforeach()
endfunction()
cmake_language(DEFER DIRECTORY "${CMAKE_SOURCE_DIR}" CALL cyber_enable_crash_reporting "${CMAKE_SOURCE_DIR}")
