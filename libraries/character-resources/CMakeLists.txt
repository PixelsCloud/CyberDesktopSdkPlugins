cmake_minimum_required(VERSION 3.26)
if(NOT TARGET CyberDesktop::CharacterProtocol)
    add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../character-protocol" "${CMAKE_CURRENT_BINARY_DIR}/protocol")
endif()
find_package(Qt6 REQUIRED COMPONENTS Core)
add_library(cd_character_resources STATIC src/ResourceTree.cpp)
add_library(CyberDesktop::CharacterResources ALIAS cd_character_resources)
target_include_directories(cd_character_resources PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(cd_character_resources PRIVATE CyberDesktop::CharacterProtocol Qt6::Core)
target_compile_features(cd_character_resources PUBLIC cxx_std_20)
target_compile_options(cd_character_resources PRIVATE /utf-8 /W4)
target_compile_definitions(cd_character_resources PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
set_target_properties(cd_character_resources PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>"
    AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
