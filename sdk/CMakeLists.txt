# Internal source includes are rooted at sdk/ (tools/, support/, tests/).
# This root is private and never enters the installed public interface.
get_property(cd_sdk_local_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
foreach(target IN LISTS cd_sdk_local_targets)
    get_target_property(type ${target} TYPE)
    if(type STREQUAL "STATIC_LIBRARY" OR type STREQUAL "SHARED_LIBRARY" OR type STREQUAL "EXECUTABLE")
        target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    endif()
endforeach()
