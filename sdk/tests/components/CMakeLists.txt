cmake_minimum_required(VERSION 3.21)
project(CyberDesktopSDKComponentProbe LANGUAGES NONE)
if(NOT SDK_PREFIX)
  message(FATAL_ERROR "SDK_PREFIX must name the installed developer kit")
endif()
set(CMAKE_PREFIX_PATH "${SDK_PREFIX}")
set(CMAKE_DISABLE_FIND_PACKAGE_CyberCanvas TRUE)
find_package(CyberDesktopSDK 1.4 CONFIG REQUIRED COMPONENTS ABI PackageRuntime)
if(TARGET CyberCanvas::CyberCanvas OR TARGET CyberDesktopSDK::Rendering)
  message(FATAL_ERROR "ABI/package import unexpectedly loaded rendering targets")
endif()
get_target_property(abi_links CyberDesktopSDK::ABI INTERFACE_LINK_LIBRARIES)
get_target_property(runtime_links CyberDesktopSDK::PackageRuntime INTERFACE_LINK_LIBRARIES)
if(abi_links OR NOT runtime_links STREQUAL "CyberDesktopSDK::ABI")
  message(FATAL_ERROR "Unexpected ABI/package dependencies: ${abi_links}; ${runtime_links}")
endif()
file(WRITE "${CMAKE_BINARY_DIR}/result.txt" "ABI and PackageRuntime import without CyberCanvas, Qt or layout dependencies.\n")
