set(CD_CUBISM_ROOT "" CACHE PATH "External Cubism Native 5-r.5 SDK root (not redistributed by the SDK)")
if(NOT EXISTS "${CD_CUBISM_ROOT}/Core/include/Live2DCubismCore.h")
  message(FATAL_ERROR "Set CD_CUBISM_ROOT to the separately obtained Cubism Native 5-r.5 SDK")
endif()
add_library(cd_cubism_core STATIC IMPORTED)
set_target_properties(cd_cubism_core PROPERTIES
  IMPORTED_LOCATION "${CD_CUBISM_ROOT}/Core/lib/windows/x86_64/143/Live2DCubismCore_MT.lib"
  INTERFACE_INCLUDE_DIRECTORIES "${CD_CUBISM_ROOT}/Core/include")
set(FRAMEWORK_SOURCE D3D11)
add_library(cd_live2d_shader_cache STATIC ShaderCache.cpp)
target_link_libraries(cd_live2d_shader_cache PUBLIC Qt6::Core d3dcompiler)
target_compile_definitions(cd_live2d_shader_cache PRIVATE NOMINMAX)
set_target_properties(cd_live2d_shader_cache PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreaded AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
add_subdirectory("${CD_CUBISM_ROOT}/Framework" "${CMAKE_CURRENT_BINARY_DIR}/framework")
target_link_libraries(Framework PUBLIC cd_cubism_core cd_live2d_shader_cache d3d11 d3dcompiler)
target_compile_options(Framework PRIVATE "/FI${CMAKE_CURRENT_SOURCE_DIR}/ShaderCache.h")
set_target_properties(Framework PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreaded AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
target_compile_definitions(Framework PUBLIC NOMINMAX)
add_library(cd_live2d_protocol STATIC Protocol.cpp)
target_link_libraries(cd_live2d_protocol PUBLIC Qt6::Core)
add_library(cd_live2d_model STATIC Assets.cpp Model.cpp)
target_link_libraries(cd_live2d_model PUBLIC Qt6::Gui Framework)
add_library(cd_live2d_worker_process STATIC WorkerProcess.cpp)
target_link_libraries(cd_live2d_worker_process PUBLIC Qt6::Core)
add_library(cd_live2d_session STATIC Session.cpp)
target_link_libraries(cd_live2d_session PUBLIC Qt6::Core PRIVATE cd_live2d_worker_process cd_live2d_protocol Qt6::Network Qt6::Gui d3d11 dxgi)
foreach(target cd_live2d_protocol cd_live2d_model cd_live2d_session cd_live2d_worker_process)
  target_include_directories(${target} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../..")
  target_compile_definitions(${target} PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
  target_compile_options(${target} PRIVATE /utf-8 /W4)
  set_target_properties(${target} PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreaded AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
endforeach()
add_executable(cyber_live2d_host Worker.cpp)
add_executable(cd_live2d_fault_worker EXCLUDE_FROM_ALL Worker.cpp)
target_compile_definitions(cd_live2d_fault_worker PRIVATE CD_LIVE2D_TEST_WORKER)
foreach(target cyber_live2d_host cd_live2d_fault_worker)
  target_link_libraries(${target} PRIVATE cd_live2d_model cd_live2d_protocol Qt6::Network dxgi)
  target_compile_options(${target} PRIVATE /utf-8 /W4)
  set_target_properties(${target} PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreaded AUTOMOC OFF AUTOUIC OFF AUTORCC OFF
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
endforeach()
add_custom_command(TARGET cyber_live2d_host POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CD_CUBISM_ROOT}/Framework/src/Rendering/D3D11/Shaders"
    "${CMAKE_BINARY_DIR}/live2d/FrameworkShaders")
