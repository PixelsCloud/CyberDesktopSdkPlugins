cmake_minimum_required(VERSION 3.26)
project(SdkInstalledConsumer LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_DISABLE_PRECOMPILE_HEADERS ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
enable_testing()

if(CD_TEST_CORE_ONLY)
    find_package(CyberDesktopSDK CONFIG REQUIRED COMPONENTS Core)
    if(TARGET CyberCanvas::CyberCanvas OR TARGET CyberDesktopSDK::Rendering)
        message(FATAL_ERROR "Core unexpectedly loads rendering dependencies")
    endif()
    add_executable(core_consumer Core.cpp)
    target_link_libraries(core_consumer PRIVATE CyberDesktopSDK::Core)
    target_compile_options(core_consumer PRIVATE /utf-8 /W4)
    add_test(NAME core_consumer COMMAND core_consumer)
    return()
endif()

find_package(CyberDesktopSDK CONFIG REQUIRED COMPONENTS SDK)
if(NOT TARGET CyberDesktopSDK::Tool)
    message(FATAL_ERROR "Installed SDK does not export its packaging tool")
endif()
get_target_property(tool CyberDesktopSDK::Tool IMPORTED_LOCATION)
cmake_path(IS_PREFIX CyberDesktopSDK_ROOT "${tool}" NORMALIZE local_tool)
if(NOT local_tool)
    message(FATAL_ERROR "Tool escaped the relocated SDK")
endif()

# Every installed public header must compile without a prefix header or PCH.
file(GLOB_RECURSE headers RELATIVE "${CyberDesktopSDK_ROOT}/include"
    "${CyberDesktopSDK_ROOT}/include/sdk/*.h")
set(header_units "")
foreach(header IN LISTS headers)
    string(MAKE_C_IDENTIFIER "${header}" unit)
    set(source "${CMAKE_CURRENT_BINARY_DIR}/headers/${unit}.cpp")
    file(WRITE "${source}" "#include <${header}>\n")
    list(APPEND header_units "${source}")
endforeach()
add_library(header_contract OBJECT ${header_units})
target_link_libraries(header_contract PRIVATE CyberDesktopSDK::SDK)
target_compile_options(header_contract PRIVATE /utf-8 /W4)

set(CD_CLOCK_BUILD_TESTS OFF CACHE BOOL "Only selected clock acceptance")
foreach(plugin clock-suite weather-suite air-quality-suite)
    add_subdirectory("${CD_PUBLIC_SOURCE}/plugins/${plugin}" "plugins/${plugin}")
endforeach()
add_test(NAME weather_display COMMAND cd_weather_tests "${CD_PUBLIC_SOURCE}/plugins/weather-suite" "${CMAKE_CURRENT_BINARY_DIR}/Testing/weather")
add_test(NAME air_display COMMAND cd_air_tests "${CD_PUBLIC_SOURCE}/plugins/air-quality-suite" "${CMAKE_CURRENT_BINARY_DIR}/Testing/air")
